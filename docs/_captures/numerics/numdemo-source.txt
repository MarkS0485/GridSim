// Build the Jacobian exactly as the production engines hold it: the structural pattern from
// SparseNewtonRaphsonPowerFlow.BuildJacobianStructure, values copied in from the captured dense Jacobian.
CsrMatrix ProductionJacobian(GridSim.Core.Model.GridModel model, SolveTrace tr, double[,] jdense)
{
    int nb = model.BusCount;
    var index = new Dictionary<int, int>(nb);
    for (int i = 0; i < nb; i++) index[model.Buses[i].Id] = i;
    Type spT = typeof(SparseLu).Assembly.GetType("GridSim.Core.PowerFlow.SparseNewtonRaphsonPowerFlow")!;
    var y = spT.GetMethod("BuildY", NP)!.Invoke(null, new object[] { model, index })!;
    int[] pPos = Enumerable.Repeat(-1, nb).ToArray(), qPos = Enumerable.Repeat(-1, nb).ToArray();
    int[] pvpq = tr.Passes[0].PvPqIndices, pq = tr.Passes[0].PqIndices;
    for (int k = 0; k < pvpq.Length; k++) pPos[pvpq[k]] = k;
    for (int k = 0; k < pq.Length; k++) qPos[pq[k]] = pvpq.Length + k;
    int mm = pvpq.Length + pq.Length;
    var slots = new int[4 * ((ComplexCsr)y).RowPtr[nb]];
    var jac = (CsrMatrix)spT.GetMethod("BuildJacobianStructure", NP)!.Invoke(null, new object[] { y, pPos, qPos, mm, slots })!;
    for (int r = 0; r < mm; r++)
        for (int k = jac.RowPtr[r]; k < jac.RowPtr[r + 1]; k++) jac.Val[k] = jdense[r, jac.ColIdx[k]];
    return jac;
}
int NonZero(CsrMatrix mat) { int c = 0; foreach (double v in mat.Val) if (v != 0.0) c++; return c; }

int Bandwidth(CsrMatrix mat, int[] perm)
{
    var pos = new int[mat.N]; for (int k = 0; k < mat.N; k++) pos[perm[k]] = k;
    int bw = 0;
    for (int i = 0; i < mat.N; i++) for (int k = mat.RowPtr[i]; k < mat.RowPtr[i + 1]; k++) bw = Math.Max(bw, Math.Abs(pos[i] - pos[mat.ColIdx[k]]));
    return bw;
}
int[] ident = Enumerable.Range(0, m14).ToArray();
int Fill(CsrMatrix mat, int[] perm)
{
    object fa = luT.GetMethod("FactorWithOrdering", NP)!.Invoke(null, new object[] { mat, perm })!;
    var t = fa.GetType();
    var l = (int[])t.GetField("_lp", NP)!.GetValue(fa)!; var u = (int[])t.GetField("_up", NP)!.GetValue(fa)!;
    return l[mat.N] + u[mat.N] - mat.N;
}

// ---- Part 3: the two builders on a row with many stamps ------------------------------------------
Console.WriteLine();
// Search deterministic random rows of 20 stamps (5 distinct columns, many duplicates) for one where the
// real builder's unstable sort changes the summation order of some cell.
for (int seed = 0; seed < 100000; seed++)
{
    var rg = new Random(seed);
    var st = new List<(int col, double v)>();
    for (int k = 0; k < 20; k++) st.Add((rg.Next(0, 5), (rg.NextDouble() - 0.5) * Math.Pow(10, rg.Next(-8, 9))));
    var rb = new CsrMatrix.Builder(5); var cb = new ComplexCsr.Builder(5);
    foreach (var (col, v) in st) { rb.Add(0, col, v); cb.Add(0, col, v, 0.0); }
    CsrMatrix rm = rb.Build(); ComplexCsr cm = cb.Build();
    // reference: insertion-order sum per column, starting from +0.0
    var refv = new double[5]; var seen = new bool[5];
    foreach (var (col, v) in st) { refv[col] += v; seen[col] = true; }
    bool differs = false;
    for (int k = rm.RowPtr[0]; k < rm.RowPtr[1]; k++)
        if (BitConverter.DoubleToInt64Bits(rm.Val[k]) != BitConverter.DoubleToInt64Bits(refv[rm.ColIdx[k]])) differs = true;
    bool cdiff = false;
    for (int k = cm.RowPtr[0]; k < cm.RowPtr[1]; k++)
        if (BitConverter.DoubleToInt64Bits(cm.G[k]) != BitConverter.DoubleToInt64Bits(refv[cm.ColIdx[k]])) cdiff = true;
    if (cdiff) { Console.WriteLine($"UNEXPECTED: ComplexCsr differs at seed {seed}"); break; }
    if (differs)
    {
        Console.WriteLine($"seed {seed}: row 0 with 20 stamps over 5 columns; real CsrMatrix.Builder differs from insertion-order sum");
        Console.WriteLine("col  insertion-order sum (bits)    ComplexCsr.Builder (bits)     CsrMatrix.Builder (bits)");
        for (int k = rm.RowPtr[0]; k < rm.RowPtr[1]; k++)
        {
            int c = rm.ColIdx[k];
            Console.WriteLine($"{c,3}  {BitConverter.DoubleToInt64Bits(refv[c]):X16}  {BitConverter.DoubleToInt64Bits(cm.G[k]):X16}  {BitConverter.DoubleToInt64Bits(rm.Val[k]):X16}");
        }
        break;
    }
    if (seed == 99999) Console.WriteLine("no differing row found in 100000 seeds");
}
